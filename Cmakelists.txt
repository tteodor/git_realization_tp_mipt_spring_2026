cmake_minimum_required(VERSION 3.28)
project(gitpta)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(
    gitpta
    main.cpp
    init.cpp
    add.cpp
    commit.cpp
    status.cpp
    show.cpp
    switch.cpp
    pure_merge.cpp
    ignore.cpp
    log.cpp
    )

install(TARGETS gitpta DESTINATION bin)
install(FILES gitpta-completion.bash DESTINATION share/bash-completion/completions RENAME gitpta)
install(FILES gitpta.1 DESTINATION share/man/man1)

include(CPack)
set(CPACK_GENERATOR "DEB")
set(CPACK_PACKAGE_NAME "gitpta")
set(CPACK_PACKAGE_VERSION "1.0.0")
set(CPACK_DEBIAN_PACKAGE_DESCRIPTION "custom git, pta mipt project")
